# [0.0.1 T1] specs/features/0.0.1-shell.md
cmake_minimum_required(VERSION 3.28)
project(KGroundControl VERSION 0.0.2 LANGUAGES CXX
    HOMEPAGE_URL "https://github.com/jayceleedev/k-groundcontrol")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 LinguistTools Test QuickTest QuickEffects Graphs Svg)
qt_standard_project_setup(REQUIRES 6.5)
include(CTest)
set(KGC_EXTRA_TESTS_DIR "" CACHE PATH "외부 시험 폴더")
include(cmake/gen/SettingsGen.cmake)
foreach(module app/errors app/log app/paths app/boot app/settings app/i18n ui)
    add_subdirectory(src/platform/${module})
endforeach()
foreach(module shell appSettings)
    add_subdirectory(src/features/${module})
endforeach()
add_subdirectory(src/main)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(KGC_EXTRA_TESTS_DIR)
    add_subdirectory("${KGC_EXTRA_TESTS_DIR}" extra_tests)
endif()
