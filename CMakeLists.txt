# [0.0.1 T1] specs/features/0.0.1-shell.md
cmake_minimum_required(VERSION 3.28)
project(KGroundControl VERSION 0.0.1 LANGUAGES CXX
    HOMEPAGE_URL "https://github.com/jayceleedev/k-groundcontrol")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 LinguistTools Test QuickTest)
qt_standard_project_setup(REQUIRES 6.5)
include(CTest)
set(KGC_EXTRA_TESTS_DIR "" CACHE PATH "외부 시험 폴더")
# 모듈은 순서대로 추가되는 동안에도 각 단계의 빌드를 확인할 수 있게 해요.
if(EXISTS "${PROJECT_SOURCE_DIR}/cmake/gen/SettingsGen.cmake")
    include(cmake/gen/SettingsGen.cmake)
endif()
foreach(module app/errors app/log app/paths app/boot app/settings app/i18n ui)
    if(EXISTS "${PROJECT_SOURCE_DIR}/src/platform/${module}/CMakeLists.txt")
        add_subdirectory(src/platform/${module})
    endif()
endforeach()
foreach(module shell appSettings)
    if(EXISTS "${PROJECT_SOURCE_DIR}/src/features/${module}/CMakeLists.txt")
        add_subdirectory(src/features/${module})
    endif()
endforeach()
if(EXISTS "${PROJECT_SOURCE_DIR}/src/main/CMakeLists.txt")
    add_subdirectory(src/main)
endif()
if(EXISTS "${PROJECT_SOURCE_DIR}/tests/CMakeLists.txt")
    add_subdirectory(tests)
endif()
if(KGC_EXTRA_TESTS_DIR)
    add_subdirectory("${KGC_EXTRA_TESTS_DIR}" extra_tests)
endif()
